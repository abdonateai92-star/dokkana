/* ضغط الصور على جهاز الأدمن قبل الحفظ: أقصى بُعد 900px، JPEG 75% */
function compressImage(file){
  return new Promise((res, rej) => {
    const img = new Image();
    img.onload = () => {
      const max = 900;
      let w = img.width, h = img.height;
      if (Math.max(w,h) > max) { const r = max/Math.max(w,h); w = Math.round(w*r); h = Math.round(h*r); }
      const cv = document.createElement("canvas"); cv.width=w; cv.height=h;
      cv.getContext("2d").drawImage(img, 0, 0, w, h);
      URL.revokeObjectURL(img.src);
      res(cv.toDataURL("image/jpeg", 0.75));
    };
    img.onerror = rej;
    img.src = URL.createObjectURL(file);
  });
}
async function handleFiles(files){
  $("prMsg").textContent = "جاري تجهيز الصور…";
  for (const f of Array.from(files).slice(0, 3 - editImages.length)) {
    try {
      const dataUrl = await compressImage(f);
      if (dataUrl.length > 900000) { $("prMsg").textContent = "صورة كبيرة حتى بعد الضغط — جرب صورة أصغر"; continue; }
      editImages.push(dataUrl);
    } catch(e){ $("prMsg").textContent = "في صورة مااتقرتش صح"; }
  }
  $("prMsg").textContent = editImages.length ? editImages.length + " صورة جاهزة ✅" : "";
  renderImgsPreview();
  $("prFiles").value = "";
}
function renderImgsPreview(){
  const el = $("prImgsPreview"); if (!el) return;
  el.innerHTML = editImages.map((s,i)=>`<div class="im"><img src="${s}"><button class="x" onclick="editImages.splice(${i},1);renderImgsPreview()">✕</button></div>`).join("");
}
async function startEdit(id){
  const p = (await db.collection("products").doc(id).get()).data(); if(!p) return;
  editId = id;
  editImages = (p.images||[]).filter(s=>s.startsWith("data:"));
  renderProducts();
  $("prName").value = p.name||""; $("prPrice").value = p.price||""; $("prOld").value = p.oldPrice||"";
  $("prStock").value = (p.stock===null||p.stock===undefined) ? "" : p.stock;
  $("prCat").value = p.categoryId||""; $("prDesc").value = p.description||"";
  $("prImgs").value = (p.images||[]).filter(s=>!s.startsWith("data:")).join(", ");
  $("prSizes").value = (p.sizes||[]).join(", "); $("prColors").value = (p.colors||[]).join(", ");
  $("prActive").checked = p.active!==false;
  window.scrollTo(0,0);
}
async function saveProduct(){
  const data = {
    name: $("prName").value.trim(), price: Number($("prPrice").value||0),
    oldPrice: Number($("prOld").value||0) || null, categoryId: $("prCat").value,
    description: $("prDesc").value.trim(),
    images: editImages.concat(split($("prImgs").value)).slice(0,3),
    sizes: split($("prSizes").value), colors: split($("prColors").value),
    stock: $("prStock").value === "" ? null : Number($("prStock").value),
    active: $("prActive").checked
  };
  if (!data.name || !data.price) { $("prMsg").textContent = "اكتب اسم المنتج وسعره على الأقل"; return; }
  if (!data.images.length) { $("prMsg").textContent = "ضيف صورة واحدة على الأقل"; return; }
  $("prMsg").textContent = "جاري الحفظ…";
  try {
    if (editId) await db.collection("products").doc(editId).update(data);
    else { data.createdAt = firebase.firestore.FieldValue.serverTimestamp(); await db.collection("products").add(data); }
    editId = null; editImages = [];
    renderProducts();
  } catch(e){ $("prMsg").textContent = "حصل خطأ أثناء الحفظ — لو الصور كتيرة قللها"; }
}
async function delProduct(id){ if(confirm("حذف المنتج نهائيًا؟")) { await db.collection("products").doc(id).delete(); renderProducts(); } }

/* ============ الأقسام ============ */
async function renderCats(){
  const cats = await fetchCats();
  $("main").innerHTML = `
    <div class="card">
      <h3>➕ إضافة قسم</h3>
      <div class="row">
        <div class="field" style="flex:2"><label>اسم القسم</label><input id="cName" placeholder="مثال: ملابس رجالي"></div>
        <div class="field" style="flex:1"><label>الترتيب</label><input id="cOrder" type="number" value="${cats.length+1}"></div>
      </div>
      <button class="btn gold" onclick="saveCat()">إضافة القسم</button>
    </div>
    <div class="card"><h3>الأقسام (${cats.length})</h3>
      ${cats.length ? cats.map(c=>`
        <div class="item">
          <div><b>${c.name}</b> <span class="muted">ترتيب: ${c.order||0} ${c.active===false?"— (مخفي)":""}</span></div>
          <div class="row">
            <button class="btn gray" onclick="toggleCat('${c.id}', ${c.active!==false})">${c.active!==false?"إخفاء":"إظهار"}</button>
            <button class="btn red" onclick="delCat('${c.id}')">حذف</button>
          </div>
        </div>`).join("") : `<div class="empty">أضف أول قسم (مثال: ملابس رجالي، ملابس حريمي، إكسسوارات، إلكترونيات)</div>`}
    </div>`;
}
async function saveCat(){
  const name = $("cName").value.trim(); if(!name) return alert("اكتب اسم القسم");
  await db.collection("categories").add({name, order:Number($("cOrder").value||0), active:true});
  renderCats();
}
async function toggleCat(id, cur){ await db.collection("categories").doc(id).update({active:!cur}); renderCats(); }
async function delCat(id){ if(confirm("حذف القسم؟ (المنتجات اللي فيه هتفضل بس من غير قسم)")) { await db.collection("categories").doc(id).delete(); renderCats(); } }

async function renderPromos(){
  const s = await db.collection("promos").get();
  const promos = s.docs.map(d=>({id:d.id, ...d.data()}));
  $("main").innerHTML = `
    <div class="card">
      <h3>➕ كود خصم جديد</h3>
      <div class="row">
        <div class="field" style="flex:1"><label>الكود</label><input id="pmCode" placeholder="SALE10" style="text-transform:uppercase"></div>
        <div class="field" style="flex:1"><label>النوع</label><select id="pmType"><option value="percent">نسبة %</option><option value="fixed">مبلغ ثابت</option></select></div>
      </div>
      <div class="row">
        <div class="field" style="flex:1"><label>القيمة</label><input id="pmValue" type="number" min="0" placeholder="10"></div>
        <div class="field" style="flex:1"><label>حد أدنى للطلب (اختياري)</label><input id="pmMin" type="number" min="0"></div>
        <div class="field" style="flex:1"><label>ينتهي في (اختياري)</label><input id="pmExp" type="date"></div>
      </div>
      <button class="btn gold" onclick="savePromo()">إنشاء الكود</button>
      <p class="muted" id="pmMsg" style="margin-top:8px"></p>
    </div>
    <div class="card"><h3>الأكواد (${promos.length})</h3>
      ${promos.length ? promos.map(p=>`
        <div class="item">
          <div><b>${p.id}</b> <span class="muted">${p.type==="percent" ? p.value+"%" : C.eg(p.value)}${p.minTotal?(" — فوق "+C.eg(p.minTotal)):""}${p.expiresAt?(" — ينتهي "+new Date(p.expiresAt).toLocaleDateString("ar-EG")):""}</span>
            ${p.active===false?'<span class="badge b-canceled">موقوف</span>':'<span class="badge b-delivered">شغال</span>'}</div>
          <div class="row">
            <button class="btn gray" onclick="togglePromo('${p.id}', ${p.active!==false})">${p.active!==false?"إيقاف":"تشغيل"}</button>
            <button class="btn red" onclick="delPromo('${p.id}')">حذف</button>
          </div>
        </div>`).join("") : `<div class="empty">مفيش أكواد خصم لسه</div>`}
    </div>`;
}
async function savePromo(){
  const code = $("pmCode").value.trim().toUpperCase();
  if (!/^[A-Z0-9]{3,20}$/.test(code)) { $("pmMsg").textContent = "الكود: حروف إنجليزي وأرقام (3-20)"; return; }
  const value = Number($("pmValue").value||0);
  if (!value) { $("pmMsg").textContent = "اكتب قيمة الخصم"; return; }
  const data = { type: $("pmType").value, value, minTotal: Number($("pmMin").value||0) || null,
    expiresAt: $("pmExp").value || null, active: true };
  await db.collection("promos").doc(code).set(data);
  $("pmMsg").textContent = "تم إنشاء الكود ✅"; renderPromos();
}
async function togglePromo(id, cur){ await db.collection("promos").doc(id).update({active:!cur}); renderPromos(); }
async function delPromo(id){ if(confirm("حذف الكود؟")) { await db.collection("promos").doc(id).delete(); renderPromos(); } }

      <div class="field"><label>السعر الافتراضي (ج.م) — 0 = مجاني</label><input id="sFeeDef" type="number" min="0" value="${fees.default||0}"></div>
      <p class="muted" style="margin-bottom:8px">سعر خاص لكل محافظة (سيبه فاضي = الافتراضي):</p>
      <div class="govgrid">
        ${C.GOVS.map(g=>`<div class="field"><label>${g}</label><input data-gov="${g}" type="number" min="0" placeholder="${fees.default||0}" value="${fees.govs&&fees.govs[g]!==undefined?fees.govs[g]:""}"></div>`).join("")}
      </div>
      <button class="btn gold" style="margin-top:12px" onclick="saveSettings()">حفظ الإعدادات</button>
      <p class="muted" id="sMsg" style="margin-top:8px"></p>
    </div>`;

async function saveSettings(){
  const govs = {};
  document.querySelectorAll("[data-gov]").forEach(i => { if (i.value !== "") govs[i.dataset.gov] = Number(i.value); });
  await db.collection("settings").doc("store").set({
    vodafoneNumber: $("sVf").value.trim(),
    whatsappNumber: $("sWa").value.trim(),
    announcement: $("sAnn").value.trim(),
    courierName: $("sCourier").value.trim(),
    fees: { default: Number($("sFeeDef").value||0), govs }
  }, {merge:true});
  $("sMsg").textContent = "تم الحفظ ✅ المتجر هيتحدث خلال 5 دقايق عند الزوار (أو فورًا لأي زيارة جديدة)";
}
