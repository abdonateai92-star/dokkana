<script src="firebase-config.js"></script>
<script>
firebase.initializeApp(firebaseConfig);
const db = firebase.firestore();

const GOVS = ["القاهرة","الجيزة","الإسكندرية","الدقهلية","الشرقية","الغربية","المنوفية","القليوبية","البحيرة","كفر الشيخ","دمياط","بورسعيد","الإسماعيلية","السويس","الفيوم","بني سويف","المنيا","أسيوط","سوهاج","قنا","الأقصر","أسوان","البحر الأحمر","مطروح","شمال سيناء","جنوب سيناء","الوادي الجديد"];
let settings = { vodafoneNumber:"", whatsappNumber:"", deliveryFee:0 };
let cats = [], products = [], cart = [];
let cur = null, curQty = 1, curCat = "all";

try { cart = JSON.parse(localStorage.getItem("dokkana_cart") || "[]"); } catch(e) { cart = []; }

const eg = n => Number(n||0).toLocaleString("en-EG") + " ج.م";
const $ = id => document.getElementById(id);

// ---------- تحميل البيانات ----------
async function loadAll() {
  try { const s = await db.collection("settings").doc("store").get(); if (s.exists) settings = {...settings, ...s.data()}; } catch(e){}
  const c = await db.collection("categories").where("active","==",true).get();
  cats = c.docs.map(d => ({id:d.id, ...d.data()})).sort((a,b)=>(a.order||0)-(b.order||0));
  const p = await db.collection("products").where("active","==",true).get();
  products = p.docs.map(d => ({id:d.id, ...d.data()}));
  renderCats(); renderGrid(); renderCartCount();
}

function renderCats() {
  $("cats").innerHTML = `<button class="chip ${curCat==='all'?'on':''}" onclick="pickCat('all')">الكل</button>` +
    cats.map(c => `<button class="chip ${curCat===c.id?'on':''}" onclick="pickCat('${c.id}')">${c.name}</button>`).join("");
}
function pickCat(id){ curCat = id; renderCats(); renderGrid(); }

function renderGrid() {
  const list = curCat==="all" ? products : products.filter(p => p.categoryId===curCat);
  if (!list.length) { $("grid").innerHTML = `<div class="empty">لا توجد منتجات هنا بعد — جرب قسمًا آخر 🙌</div>`; return; }
  $("grid").innerHTML = list.map(p => `
    <div class="card" onclick="openProduct('${p.id}')">
      <img src="${(p.images&&p.images[0])||''}" alt="" onerror="this.src='';this.style.background='#eee'">
      <div class="info">
        <div class="name">${p.name}</div>
        <div><span class="price">${eg(p.price)}</span>${p.oldPrice?`<span class="old">${eg(p.oldPrice)}</span>`:""}</div>
        <button class="addbtn" onclick="event.stopPropagation();openProduct('${p.id}')">أضف للسلة</button>
      </div>
    </div>`).join("");
}

// ---------- نافذة المنتج ----------
function openProduct(id) {
  cur = products.find(p => p.id===id); if(!cur) return;
  curQty = 1; $("pQty").textContent = 1;
  $("pImg").src = (cur.images&&cur.images[0])||"";
  $("pName").textContent = cur.name;
  $("pPrice").textContent = eg(cur.price);
  $("pOld").textContent = cur.oldPrice ? eg(cur.oldPrice) : "";
  $("pDesc").textContent = cur.description || "";
  if (cur.sizes && cur.sizes.length) { $("sizeRow").style.display="block"; $("pSize").innerHTML = cur.sizes.map(s=>`<option>${s}</option>`).join(""); }
  else $("sizeRow").style.display="none";
  if (cur.colors && cur.colors.length) { $("colorRow").style.display="block"; $("pColor").innerHTML = cur.colors.map(s=>`<option>${s}</option>`).join(""); }
  else $("colorRow").style.display="none";
  $("pOverlay").classList.add("show");
}
function chQty(d){ curQty = Math.max(1, curQty+d); $("pQty").textContent = curQty; }
function addToCart() {
  const size = $("sizeRow").style.display!=="none" ? $("pSize").value : "";
  const color = $("colorRow").style.display!=="none" ? $("pColor").value : "";
  const key = cur.id+"|"+size+"|"+color;
  const found = cart.find(i => i.key===key);
  if (found) found.qty += curQty;
  else cart.push({key, id:cur.id, name:cur.name, price:cur.price, image:(cur.images&&cur.images[0])||"", size, color, qty:curQty});
  saveCart(); closeOverlay("pOverlay"); openCart();
}

// ---------- السلة ----------
function saveCart(){ localStorage.setItem("dokkana_cart", JSON.stringify(cart)); renderCartCount(); }
function renderCartCount(){ $("cartCount").textContent = cart.reduce((a,i)=>a+i.qty,0); }
function cartSubtotal(){ return cart.reduce((a,i)=>a+i.price*i.qty,0); }
function openCart(){ renderCart(); $("cOverlay").classList.add("show"); }
function renderCart() {
  if (!cart.length) { $("cartItems").innerHTML = `<div class="empty">السلة فاضية 🛒</div>`; $("cartTotals").innerHTML=""; $("checkoutBtn").style.display="none"; return; }
  $("checkoutBtn").style.display="block";
  $("cartItems").innerHTML = cart.map((i,ix) => `
    <div class="cartrow">
      <img src="${i.image}" onerror="this.src=''">
      <div class="cinfo"><b>${i.name}</b><br>${i.size?("مقاس: "+i.size+" "):""}${i.color?("لون: "+i.color):""}<br>
        ${eg(i.price)} × ${i.qty}
        <div class="qty" style="margin-top:4px">
          <button onclick="cartQty(${ix},1)">+</button><b>${i.qty}</b><button onclick="cartQty(${ix},-1)">−</button>
        </div>
      </div>
      <button class="rm" onclick="cartRm(${ix})">حذف ✕</button>
    </div>`).join("");
  $("cartTotals").innerHTML = totalsHtml();
}
function totalsHtml() {
  const sub = cartSubtotal(), fee = Number(settings.deliveryFee||0);
  return `<div><span>المنتجات</span><span>${eg(sub)}</span></div>
    <div><span>الشحن</span><span>${fee?eg(fee):"يتحدد عند التأكيد"}</span></div>
    <div class="grand"><span>الإجمالي</span><span>${eg(sub+fee)}</span></div>`;
}
function cartQty(ix,d){ cart[ix].qty = Math.max(1, cart[ix].qty+d); saveCart(); renderCart(); }
function cartRm(ix){ cart.splice(ix,1); saveCart(); renderCart(); }

// ---------- إتمام الطلب ----------
function openCheckout() {
  if (!cart.length) return;
  closeOverlay("cOverlay");
  $("fGov").innerHTML = GOVS.map(g=>`<option>${g}</option>`).join("");
  $("orderTotals").innerHTML = totalsHtml();
  updateVfBox();
  $("oOverlay").classList.add("show");
}
function updateVfBox() {
  const vf = document.querySelector('input[name=pay]:checked').value === "vodafone";
  $("payCod").classList.toggle("sel", !vf);
  $("payVf").classList.toggle("sel", vf);
  $("vfBox").style.display = vf ? "block" : "none";
  if (vf) $("vfBox").innerHTML = `📱 حوّل <b>${eg(cartSubtotal()+Number(settings.deliveryFee||0))}</b> على رقم فودافون كاش:<br>
    <b style="font-size:18px">${settings.vodafoneNumber || "(سيظهر الرقم قريبًا)"}</b><br>
    بعد التحويل هنكلمك على واتساب لتأكيد الطلب.`;
}
document.querySelectorAll('input[name=pay]').forEach(r => r.addEventListener("change", updateVfBox));

async function placeOrder() {
  const name = $("fName").value.trim(), phone = $("fPhone").value.trim(), addr = $("fAddr").value.trim();
  if (!name) return alert("اكتب اسمك من فضلك");
  if (!/^01[0-9]{9}$/.test(phone)) return alert("اكتب رقم موبايل مصري صحيح (11 رقم يبدأ بـ 01)");
  if (!addr) return alert("اكتب العنوان بالتفصيل");
  const btn = $("placeBtn"); btn.disabled = true; btn.textContent = "جاري إرسال الطلب…";
  try {
    const payment = document.querySelector('input[name=pay]:checked').value;
    const order = {
      items: cart.map(i => ({productId:i.id, name:i.name, price:i.price, qty:i.qty, size:i.size||"", color:i.color||""})),
      subtotal: cartSubtotal(),
      deliveryFee: Number(settings.deliveryFee||0),
      total: cartSubtotal() + Number(settings.deliveryFee||0),
      customer: { name, phone, governorate:$("fGov").value, address:addr, notes:$("fNotes").value.trim() },
      payment, status: "new",
      createdAt: firebase.firestore.FieldValue.serverTimestamp()
    };
    const ref = await db.collection("orders").add(order);
    const num = ref.id.slice(0,6).toUpperCase();
    cart = []; saveCart();
    $("checkoutSheet").innerHTML = `
      <div class="success">
        <div class="check">✅</div>
        <h2>طلبك وصل يا ${name.split(" ")[0]}!</h2>
        <p style="margin:10px 0">رقم الطلب: <b>#${num}</b><br>هنتواصل معك على ${phone} لتأكيد الطلب.</p>
        ${settings.whatsappNumber ? `<a class="wbtn" href="https://wa.me/${settings.whatsappNumber}?text=${encodeURIComponent("أهلًا، أنا "+name+" — طلبي رقم #"+num)}" target="_blank">📲 تابع طلبك واتساب</a>` : ""}
        <br><br><button class="bigbtn" onclick="location.reload()">كمّل تسوق 🛍️</button>
      </div>`;
  } catch(e) {
    alert("حصل خطأ أثناء إرسال الطلب — جرب تاني");
    btn.disabled = false; btn.textContent = "تأكيد الطلب 🚀";
  }
}

function closeOverlay(id){ $(id).classList.remove("show"); }
$("cartBtn").onclick = openCart;
["pOverlay","cOverlay","oOverlay"].forEach(id =>
  $(id).addEventListener("click", e => { if (e.target.id===id) closeOverlay(id); }));

loadAll().catch(e => { $("grid").innerHTML = `<div class="empty">تعذر تحميل المنتجات — تأكد من الإنترنت وجرب تاني</div>`; });
</script>
</body>
